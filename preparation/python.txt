# *args - n number of arguments
# **kwargs - n number of named arguments like name=a, age=90
# kwargs - is a dictionary
def add_all(*numbers):
    total = 0
    for number in numbers:
        total += number
    return total

def print_user(**kwargs):
    print(kwargs)

print_user(name="John",age=40,city="New York")
